/* Ammo Making - recipes on the vanilla Build 42 stations.

   Every block is adapted from a vanilla 42.20.4 recipe.

   Metallurgy, from entities/blacksmith/craftRecipes/recipes_blacksmith_furnaces_i.txt:

     AmmoMaking_SmeltZincOre     <- SmeltCopperOre (same fields, zinc items)
     AmmoMaking_Cast*Ingot       <- CastIronIngot (tongs + mold kept, Furnace tag) with the
                                    kept empty crucible of ExtractIronFromSmallItem
     AmmoMaking_CastBrassIngots  <- same casting block, two metals in

   Case stock, from recipes_blacksmith_other_metals.txt and recipes_improvised_weapons.txt:

     AmmoMaking_ForgeSmallBrassSheets <- Forge_Copper_Sheet (tools, flags, timed action)
     AmmoMaking_PunchBrassCaseCups    <- cold work on a surface: the punch line of the
                                         scrap-armour recipes, the hammer line and timed
                                         action of NailSpikeWeapon

   The recipes live in module Base, like every vanilla recipe, so the engine
   handles them exactly as its own; the AmmoMaking_ prefix keeps the ids
   unique. Station attachment is the Tags line only; the furnaces and forges
   have no lit or fuel state, charcoal is an input.

   No SkillRequired / xpAward lines: the Ammo Making perk is registered from
   Lua after scripts are parsed, so the engine would drop them. AC_Materials
   attaches the skill requirement at boot and awards XP from OnCreate.

   AC_Materials.RECIPES mirrors this file; tests/run_tests.lua fails when
   the two differ. */

module Base
{
    craftRecipe AmmoMaking_SmeltZincOre
    {
        time = 200,
        Tags = PrimitiveFurnace,
        category = Blacksmithing,
        OnCreate = AC_Materials.onSmeltZincOre,
        inputs
        {
            item 4 tags[base:charcoal],
            item 1 [AmmoMaking.ZincOre],
        }
        outputs
        {
            item 10 AmmoMaking.ZincScrap,
        }
    }

    craftRecipe AmmoMaking_CastCopperIngot
    {
        time = 200,
        Tags = Furnace,
        category = Blacksmithing,
        OnCreate = AC_Materials.onCastCopperIngot,
        inputs
        {
            item 1 [Base.CeramicCrucible] mode:keep flags[IsEmpty],
            item 1 tags[base:crudetongs;base:tongs] mode:keep flags[MayDegradeLight],
            item 4 tags[base:charcoal],
            item 10 [Base.CopperScrap],
            item 1 [Base.ClayIngotMold;Base.IronIngotMold;Base.SteelIngotMold] mode:keep,
        }
        outputs
        {
            item 1 Base.CopperIngot,
        }
    }

    craftRecipe AmmoMaking_CastZincIngot
    {
        time = 200,
        Tags = Furnace,
        category = Blacksmithing,
        OnCreate = AC_Materials.onCastZincIngot,
        inputs
        {
            item 1 [Base.CeramicCrucible] mode:keep flags[IsEmpty],
            item 1 tags[base:crudetongs;base:tongs] mode:keep flags[MayDegradeLight],
            item 4 tags[base:charcoal],
            item 10 [AmmoMaking.ZincScrap],
            item 1 [Base.ClayIngotMold;Base.IronIngotMold;Base.SteelIngotMold] mode:keep,
        }
        outputs
        {
            item 1 AmmoMaking.ZincIngot,
        }
    }

    craftRecipe AmmoMaking_CastBrassIngots
    {
        time = 200,
        Tags = Furnace,
        category = Blacksmithing,
        OnCreate = AC_Materials.onCastBrassIngots,
        inputs
        {
            item 1 [Base.CeramicCrucible] mode:keep flags[IsEmpty],
            item 1 tags[base:crudetongs;base:tongs] mode:keep flags[MayDegradeLight],
            item 10 tags[base:charcoal],
            item 7 [Base.CopperIngot],
            item 3 [AmmoMaking.ZincIngot],
            item 1 [Base.ClayIngotMold;Base.IronIngotMold;Base.SteelIngotMold] mode:keep,
        }
        outputs
        {
            item 10 Base.BrassIngot,
        }
    }

    craftRecipe AmmoMaking_ForgeSmallBrassSheets
    {
        time = 200,
        timedAction = HammerMetalStanding,
        Tags = PrimitiveForge,
        category = Blacksmithing,
        OnCreate = AC_Materials.onForgeSmallBrassSheets,
        inputs
        {
            item 1 tags[base:charcoal],
            item 1 [Base.BrassIngot],
            item 1 tags[base:hammer;base:clubhammer] mode:keep flags[Prop1;MayDegradeLight],
            item 1 tags[base:tongs;base:metalworkingpliers] mode:keep flags[Prop2;MayDegradeLight],
        }
        outputs
        {
            item 10 AmmoMaking.SmallBrassSheet,
        }
    }

    craftRecipe AmmoMaking_PunchBrassCaseCups
    {
        time = 100,
        timedAction = MakingHammer_Surface,
        Tags = AnySurfaceCraft,
        category = Metalworking,
        OnCreate = AC_Materials.onPunchBrassCaseCups,
        inputs
        {
            item 1 [AmmoMaking.SmallBrassSheet],
            item 1 tags[base:metalworkingpunch;base:smallpunch] mode:keep flags[MayDegradeLight],
            item 1 tags[base:hammer] mode:keep flags[MayDegradeVeryLight],
        }
        outputs
        {
            item 2 AmmoMaking.BrassCaseCup,
        }
    }
}
