/* Ammo Making: Reloading Press (add-on) - the press as a placed station.

   EXPERIMENTAL, NOT YET SEEN IN GAME. This file is in the ADD-ON mod
   AmmoMakingPress, not in the main mod: a malformed entity stops every
   world from loading (docs/RELOADING_PRESS_DESIGN.md 2.4), so the station
   exists only for someone who ticks the add-on. Ticking it is the feature
   switch "reloadingPress" (AC_Features.lua in the main mod).

   It follows vanilla's Hand_Press field for field
   (media/scripts/generated/entities/pottery/workstations/entity_hand_press.txt,
   42.20.4): a CraftBench whose Recipes tag is the press's own, two faces,
   a build recipe. Its window skin is in AC_ReloadingPress_xuiSkin.txt, a
   file of its own as vanilla's is. The skill line and the tooltip follow
   vanilla's welded stations (entity_feedingtrough.txt).

   PLACEHOLDER_VISUAL: the two sprites are the add-on's own tile sheet
   (art/reloading_press/, built by tools/build_tiles.py), programmer art
   drawn from boxes. They are the add-on's own names, so no other entity
   can claim them. The window icon is vanilla's Build_Handpress. Both are
   listed in docs/PLACEHOLDER_ASSETS.md; replacing the art means replacing
   two PNG files and rebuilding the sheet, nothing here.

   The build recipe's skill is a vanilla one: the Ammo Making perk is
   registered from Lua after scripts are parsed, and the engine drops a
   skill name it does not know. Its inputs are a first guess. */

module Base
{
    entity AmmoMaking_ReloadingPress
    {
        component UiConfig
        {
            xuiSkin = default,
            entityStyle = ES_AmmoMaking_ReloadingPress,
            uiEnabled = true,
        }
        component CraftBench
        {
            Recipes = AmmoMakingReloadingPress,
        }
        component SpriteConfig
        {
            health = 100,
            skillBaseHealth = 20,
            face S
            {
                layer
                {
                    row = ammomaking_press_01_0,
                }
            }
            face E
            {
                layer
                {
                    row = ammomaking_press_01_1,
                }
            }
        }
        component CraftRecipe
        {
            timedAction = BuildWallHammer,
            time = 100,
            category = Blacksmithing,
            SkillRequired = MetalWelding:2,
            xpAward = MetalWelding:10,
            Tooltip = Tooltip_craft_AmmoMakingReloadingPressDesc,
            inputs
            {
                item 1 tags[base:hammer] mode:keep flags[Prop1;MayDegradeVeryLight],
                item 2 [Base.SteelBarHalf],
                item 4 [Base.Plank],
                item 8 [Base.Nails],
            }
        }
    }
}
